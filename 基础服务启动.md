  --name lightbot-postgres \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres \
  -e POSTGRES_DB=lightbot \
  -p 5432:5432 \
  --restart unless-stopped \
  pgvector/pgvector:pg15
